3.1 Main CI Pipeline

.github/workflows/hitloop-ci.yml
# =============================================================
# Hitloop Moodle — Continuous Integration Pipeline
# Runs on every PR and push to develop/main
# =============================================================
name: Hitloop CI

on:
  pull_request:
    branches: [develop, main]
  push:
    branches: [develop]

concurrency:
  group: ci-${{ github.ref }}
  cancel-in-progress: true

env:
  PHP_VERSION: '8.2'
  NODE_VERSION: '20'

jobs:
  # ─── JOB 1: PHP Lint & Static Analysis ───
  lint:
    name: "🔍 PHP Lint & Static Analysis"
    runs-on: ubuntu-24.04
    steps:
      - name: Checkout code with submodules
        uses: actions/checkout@v4
        with:
          submodules: recursive
          fetch-depth: 0

      - name: Setup PHP ${{ env.PHP_VERSION }}
        uses: shivammathur/setup-php@v2
        with:
          php-version: ${{ env.PHP_VERSION }}
          extensions: mysqli, redis, gd, intl, mbstring, xml, curl, zip, sodium, igbinary
          tools: phpcs, phpstan
          coverage: none

      - name: Cache Composer dependencies
        uses: actions/cache@v4
        with:
          path: vendor
          key: composer-${{ hashFiles('composer.lock') }}

      - name: Install Composer dependencies
        run: composer install --no-dev --prefer-dist --no-progress

      - name: PHP Syntax Check (custom plugins only)
        run: |
          find local/veda local/my_webservices local/ai_manager \
               mod/codingplayground mod/coursedashboard \
               -name "*.php" -not -path "*/vendor/*" \
               -exec php -l {} \;

      - name: PHPCS — Moodle Coding Standards
        run: |
          phpcs --standard=phpcs.xml --extensions=php \
            --ignore=*/vendor/*,*/node_modules/*,*/dist/* \
            local/veda/ local/my_webservices/ local/ai_manager/ \
            mod/codingplayground/ mod/coursedashboard/

  # ─── JOB 2: PHPUnit Tests ───
  phpunit:
    name: "🧪 PHPUnit Tests"
    runs-on: ubuntu-24.04
    services:
      mysql:
        image: mysql:8.4
        env:
          MYSQL_ROOT_PASSWORD: root
          MYSQL_DATABASE: moodle_test
          MYSQL_USER: moodle
          MYSQL_PASSWORD: moodle_test_pass
        ports: ['3306:3306']
        options: --health-cmd="mysqladmin ping" --health-interval=10s --tmpfs /var/lib/mysql:rw
      redis:
        image: redis:7
        ports: ['6379:6379']
    steps:
      - uses: actions/checkout@v4
        with: { submodules: recursive }
      - uses: shivammathur/setup-php@v2
        with: { php-version: '8.2', extensions: mysqli,redis,gd,intl }
      - run: composer install --prefer-dist
      - name: Prepare Moodle for PHPUnit
        run: |
          cp .github/workflows/config-template.php config.php
          mkdir -p ../moodledata ../phpunitdata
          php admin/tool/phpunit/cli/init.php --no-composer-self-update
        env: { dbtype: mysqli }
      - name: Run PHPUnit — Custom Plugins
        run: vendor/bin/phpunit --log-junit phpunit-results.xml
      - uses: actions/upload-artifact@v4
        with: { name: phpunit-results, path: phpunit-results.xml }

  # ─── JOB 3: Frontend Builds ───
  frontend:
    name: "🎨 Frontend Builds"
    runs-on: ubuntu-24.04
    steps:
      - uses: actions/checkout@v4
        with: { submodules: recursive }
      - uses: actions/setup-node@v4
        with: { node-version: '20' }
      - name: Grunt — Moodle AMD modules
        run: npm ci && npx grunt
      - name: Build Course Dashboard frontend
        working-directory: mod/coursedashboard/client
        run: npm ci && npm run build

  # ─── JOB 4: Security Scanning ───
  security:
    name: "🔒 Security Scan"
    runs-on: ubuntu-24.04
    steps:
      - uses: actions/checkout@v4
      - name: Trivy Vulnerability Scan
        uses: aquasecurity/trivy-action@master
        with:
          scan-type: 'fs'
          severity: 'CRITICAL,HIGH'

3.2 Deploy to Staging Pipeline

.github/workflows/deploy-staging.yml
name: Deploy to Staging

on:
  push:
    branches: [develop]

jobs:
  deploy-staging:
    name: "🚀 Deploy → Staging"
    runs-on: ubuntu-24.04
    environment:
      name: staging
      url: https://staging.hitloop.com
    steps:
      - uses: actions/checkout@v4
        with: { submodules: recursive }

      - name: Build production artifact
        run: |
          composer install --no-dev --optimize-autoloader
          npm ci && npx grunt
          cd mod/coursedashboard/client && npm ci && npm run build && cd ../../..
          RELEASE="release-$(date +%Y%m%d-%H%M%S)-${GITHUB_SHA::8}"
          echo "RELEASE=$RELEASE" >> $GITHUB_ENV
          tar czf /tmp/${RELEASE}.tar.gz --exclude='.git' --exclude='node_modules' \
            --exclude='config.php' --exclude='*.log' .

      - name: Deploy via SSH
        run: |
          # 1. Upload artifact to server
          scp /tmp/$RELEASE.tar.gz $DEPLOY_USER@$HOST:/tmp/
          # 2. Enable maintenance mode
          ssh $DEPLOY_USER@$HOST "php /var/www/html/moodle/admin/cli/maintenance.php --enable"
          # 3. Extract release
          ssh $DEPLOY_USER@$HOST "mkdir -p /var/www/releases/$RELEASE && \
            tar xzf /tmp/$RELEASE.tar.gz -C /var/www/releases/$RELEASE"
          # 4. Link shared config
          ssh $DEPLOY_USER@$HOST "ln -sf /var/www/shared/config.php /var/www/releases/$RELEASE/config.php"
          # 5. Atomic symlink switch
          ssh $DEPLOY_USER@$HOST "ln -sfn /var/www/releases/$RELEASE /var/www/html/moodle"
          # 6. Moodle upgrade + cache purge
          ssh $DEPLOY_USER@$HOST "php /var/www/html/moodle/admin/cli/upgrade.php --non-interactive"
          ssh $DEPLOY_USER@$HOST "php /var/www/html/moodle/admin/cli/purge_caches.php"
          # 7. Disable maintenance
          ssh $DEPLOY_USER@$HOST "php /var/www/html/moodle/admin/cli/maintenance.php --disable"

      - name: Smoke test
        run: |
          HTTP_CODE=$(curl -o /dev/null -s -w "%{http_code}" "https://staging.hitloop.com/login/index.php")
          [ "$HTTP_CODE" -eq 200 ] || exit 1

3.3 Deploy to Production Pipeline

.github/workflows/deploy-production.yml
name: Deploy to Production

on:
  push:
    tags: ['v*.*.*']

jobs:
  deploy-production:
    name: "🏭 Deploy → Production"
    runs-on: ubuntu-24.04
    environment:
      name: production  # ← requires manual approval in GitHub settings
      url: https://devlearn.hitloop.com
    steps:
      - uses: actions/checkout@v4
        with: { submodules: recursive }

      - name: Build + Deploy (same as staging, with DB backup first)
        run: |
          # Step 0: Create DB backup BEFORE deployment
          ssh $USER@$HOST "mysqldump -u root moodle --single-transaction | gzip > /var/www/backups/db-$(date +%Y%m%d).sql.gz"
          # Steps 1-7: Same as staging deployment above
          # ... (artifact upload, maintenance mode, symlink switch, upgrade, etc.)

      - name: Smoke test production
        run: |
          HTTP_CODE=$(curl -o /dev/null -s -w "%{http_code}" "https://devlearn.hitloop.com/login/index.php")
          [ "$HTTP_CODE" -eq 200 ] && echo "✅ Production OK" || exit 1

  # ─── Automatic Rollback (if smoke test fails) ───
  rollback:
    name: "⏪ Emergency Rollback"
    needs: deploy-production
    if: failure()
    runs-on: ubuntu-24.04
    steps:
      - name: Rollback to previous release
        run: |
          PREVIOUS=$(ssh $USER@$HOST "ls -dt /var/www/releases/*/ | sed -n '2p'")
          ssh $USER@$HOST "ln -sfn $PREVIOUS /var/www/html/moodle"
          ssh $USER@$HOST "php /var/www/html/moodle/admin/cli/purge_caches.php"
          ssh $USER@$HOST "php /var/www/html/moodle/admin/cli/maintenance.php --disable"
          echo "✅ Rollback complete"
